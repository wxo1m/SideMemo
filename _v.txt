302: class DownloadWorker(QThread):
303:     progress = Signal(int)
304:     finished = Signal(str)
305:     error = Signal(str)
306: 
307:     def __init__(self, download_url: str, tag_name: str, parent=None):
308:         super().__init__(parent)
309:         self.download_url = download_url
310:         self.tag_name = tag_name
311: 
312:     def run(self):
313:         try:
314:             temp_dir = tempfile.gettempdir()
315:             clean_tag = self.tag_name.lstrip("vV")
316:             target_file = os.path.join(temp_dir, f"SideMemo-Setup-{clean_tag}.exe")
317:             urls = [self.download_url]
318:             fallback_url = f"https://raw.githubusercontent.com/{GITHUB_REPO}/main/installer/SideMemo-Setup.exe"
319:             if fallback_url not in urls:
320:                 urls.append(fallback_url)
321: 
322:             success = False
323:             last_err = None
324:             for u in urls:
325:                 if not u:
326:                     continue
327:                 try:
328:                     req = urllib.request.Request(u, headers={
329:                         "User-Agent": f"SideMemo-App/{APP_VERSION}"
330:                     })
331:                     with urllib.request.urlopen(req, timeout=45) as resp, open(target_file, "wb") as f:
332:                         total_size = int(resp.headers.get("content-length", 0))
333:                         downloaded = 0
334:                         block_size = 65536
335:                         while True:
336:                             chunk = resp.read(block_size)
337:                             if not chunk:
338:                                 break
339:                             f.write(chunk)
340:                             downloaded += len(chunk)
341:                             if total_size > 0:
342:                                 self.progress.emit(int((downloaded / total_size) * 100))
343:                     if os.path.exists(target_file) and os.path.getsize(target_file) > 500000:
344:                         success = True
345:                         break
346:                 except Exception as e:
347:                     last_err = e
348: 
349:             if success:
350:                 self.finished.emit(target_file)
351:             else:
352:                 self.error.emit(str(last_err or "다운로드에 실패했습니다."))
353:         except Exception as e:
354:             self.error.emit(str(e))
